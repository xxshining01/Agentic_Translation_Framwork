## 1. วัตถุประสงค์ (Purpose)

เอกสารฉบับนี้กำหนดรายละเอียดการออกแบบเชิงวิศวกรรม (Detailed Engineering Specification) โครงสร้างข้อมูล และขั้นตอนวิธี (Algorithms) สำหรับการสกัด (Extraction), การคัดแยกความหมาย (Semantic Classification), และการฉีดคำแปล (Safe Injection) ของข้อความส่วนติดต่อผู้ใช้ (UI Strings) ที่ฝังตัวอยู่ในไฟล์ `www/js/plugins.js` ของเกม RPG Maker MV/MZ เพื่อให้สามารถทำงานร่วมกับไปป์ไลน์การแปลของ ATF ได้อย่างสมบูรณ์แบบและปลอดภัย 100%

## 2. โครงสร้างคลาสและอินเทอร์เฟซ (Class Interface Specification)

```python
from typing import List, Dict, Tuple, Optional, Any, Set, Union
import json
import re

class PluginASTParser:
    """
    โมดูลจัดการการแยกและประกอบ JavaScript Wrapper สำหรับ www/js/plugins.js
    """
    HEADER_TEMPLATE: str = (
        "// Generated by RPG Maker.\n"
        "// Do not edit this file directly.\n"
        "var $plugins =\n"
    )

    @classmethod
    def parse_plugins_file(cls, file_path: str) -> List[Dict[str, Any]]:
        """
        สกัดก้อนอาร์เรย์ $plugins ออกจากไฟล์ JavaScript
        Returns:
            List[Dict[str, Any]]: รายการของ plugin objects ทั้งหมด
        """

    @classmethod
    def serialize_plugins_file(cls, plugins_data: List[Dict[str, Any]]) -> str:
        """
        จัดรูปแบบอาร์เรย์ $plugins กลับเป็นโค้ด JavaScript ที่ถูกต้องตามไวยากรณ์เดิม
        """


class RecursiveJsonUnpacker:
    """
    โมดูลแกะข้อมูล JSON ที่ซ้อนทับกันหลายชั้นใน parameters (Nested Stringified JSON)
    """
    @classmethod
    def is_json_string(cls, val: Any) -> bool:
        """ตรวจสอบว่าค่าสตริงขึ้นต้นด้วยโครงสร้าง JSON (object หรือ array) หรือไม่"""

    @classmethod
    def unpack(cls, data: Any, current_path: List[Union[str, int]]) -> List[Tuple[List[Union[str, int]], str]]:
        """
        ท่องลึกเข้าไปในโครงสร้างข้อมูลแบบ Recursive และส่งคืนรายการ (JSON_Path, Leaf_String)
        """


class SemanticParameterClassifier:
    """
    โมดูลคัดกรองความปลอดภัย: แยกเฉพาะ Human-Readable UI Text ออกจากโค้ด, สูตร, และไอดีเกม
    """
    DISALLOWED_KEY_SUFFIXES: Set[str] = {
        ":bool", ":css", ":color", ":file", ":img", ":image",
        ":variable", ":switch", ":id", ":i", ":e"
    }
    
    SYSTEM_KEYS_BLACKLIST: Set[str] = {
        "name", "status", "description", "folder", "filename", "skin",
        "fontFace", "formula", "renderingMode", "alwaysDash", "key",
        "right click", "show on new page", "pause until visible"
    }

    JAVASCRIPT_CODE_PATTERNS: List[str] = [
        r"Graphics\.", r"Window_Base\.", r"\.prototype\.",
        r"\bfunction\b", r"=>", r"[+\-*/]\s*\d+", r"===|!==|&&|\|\|",
        r"^#[0-9a-fA-F]{3,8}$", r"^rgba?\(", r"^(?:true|false|auto|left|center|right)$"
    ]

    ALLOWED_KEY_KEYWORDS: List[str] = [
        "Text", "Title", "Name", "Command", "Format", "Label", "Message",
        "Prompt", "Read", "Cancel", "Close", "Word", "Period", "Option"
    ]

    @classmethod
    def is_translatable_ui_string(cls, key_path: List[Union[str, int]], value: str) -> Tuple[bool, str]:
        """
        วิเคราะห์ว่าคู่ (Path, Value) เหมาะสมที่จะสกัดออกมาแปลหรือไม่
        Returns:
            (is_translatable, reason)
        """


class RecursiveJsonPacker:
    """
    โมดูลฉีดคำแปลภาษาไทยและ Re-serialize โครงสร้างข้อมูลที่ซ้อนทับกลับสู่ระดับเดิม
    """
    @classmethod
    def inject(cls, root_data: Any, path: List[Union[str, int]], new_val: str) -> None:
        """
        นำค่าคำแปลใหม่ไปแทนที่ในโครงสร้างข้อมูลตามลำดับ JSON Path
        """

    @classmethod
    def repack_parameters(cls, unpacked_params: Dict[str, Any]) -> Dict[str, str]:
        """
        Re-serialize อ็อบเจกต์ภายในพารามิเตอร์ให้กลับเป็น Stringified JSON ตามมาตรฐาน RPG Maker
        """


class ATFPluginCSVBridge:
    """
    โมดูลแปลงข้อมูลระหว่าง Plugin Text Entities กับ CSV Schema 8 คอลัมน์ของ ATF
    """
    HEADER: List[str] = [
        "Original", "Initial", "Filename", "Row", "Note", "Role", "Speaker", "Displayable"
    ]

    @classmethod
    def export_to_csv(cls, entries: List[Dict[str, Any]], output_path: str) -> None:
        """ส่งออกรายการข้อความปลั๊กอินลงไฟล์ CSV"""

    @classmethod
    def load_from_csv(cls, csv_path: str) -> Dict[str, str]:
        """โหลดคำแปลจาก CSV ในรูปแมปปิ้ง {json_path_str: initial_text}"""
```

## 3. ตารางเกณฑ์การคัดแยกพารามิเตอร์ (Classification Decision Matrix & Rules)

| ลำดับการตรวจสอบ | เงื่อนไขในข้อมูล (Condition) | ผลการจำแนก (Classification) | เหตุผลทางวิศวกรรม (Engineering Rationale) |
|---|---|:---:|---|
| **1. Type Check** | ค่าของโหนดไม่ใช่ `str` หรือเป็นสตริงว่าง | **REJECT** | ไม่ใช่ข้อมูลข้อความ |
| **2. Lunatic & Code Block Guard** | มีคำว่า `lunatic mode` หรือ `eval` ใน Path หรือขึ้นต้นด้วย `//`, `/*` | **REJECT** | เป็นบล็อกโค้ดสคริปต์หรือคอมเมนต์ ไม่ใช่ข้อความสำหรับผู้เล่น |
| **3. Asset / Hardware Key Guard** | คีย์มีคำว่า `iconname`, `icon`, `image`, `skin`, `font`, `key`, `position` | **REJECT** | เป็นชื่อไฟล์ทรัพยากรเกมหรือปุ่มลัด ห้ามแปลเด็ดขาด |
| **4. Key Suffix Guard** | คีย์ลงท้ายด้วย `:bool`, `:css`, `:color`, `:file`, `:id`, `:variable`, `:switch` | **REJECT** | เป็นชนิดข้อมูลเฉพาะทางของเกม (ตัวแปร/คอนฟิก) ห้ามแปล |
| **5. Code Syntax Guard** | มีไวยากรณ์โค้ด เช่น `Graphics.`, `.prototype.`, `this.fittingHeight`, `===`, `rgba()` | **REJECT** | เป็นสูตรคำนวณ JavaScript หากแปลจะทำให้เกมแครชทันที |
| **6. Date/Time Format Guard** | ข้อความมีเฉพาะรหัสจัดรูปแบบวันเวลา เช่น `%D %MMMM %YYYY`, `%HH%:%mm` | **REJECT** | เป็น Format Specification สำหรับแสดงผลนาฬิกา |
| **7. Pure Tags / Pure Symbols** | มีเฉพาะรหัสควบคุม เช่น `\c[6]`, `\i[160]%1` หรือตัวเลขล้วน โดยไม่มีตัวอักษรมนุษย์ | **REJECT** | ไม่มีเนื้อหาภาษาธรรมชาติที่ต้องแปล |
| **8. Whitelist UI Text** | คีย์มีคำว่า `Text`, `Title`, `Name`, `Command`, `Format`, `Label`, `Quests`, `Period` | **ACCEPT** | เป็นข้อความที่ผู้เล่นมองเห็นบนจอเกมอย่างแน่นอน 100% |

## 4. รายละเอียดรูปแบบ JSON Path Notation

เพื่อให้ขั้นตอนการสกัด (Extract) และการฉีดกลับ (Inject) มีความแม่นยำระดับศัลยกรรม ทุกข้อความจะถูกบันทึกที่อยู่ลงในคอลัมน์ `Note` ของ CSV โดยใช้ไวยากรณ์จุดและวงเล็บ (Dot & Bracket Notation):

### ตัวอย่างรูปแบบจริง:
1. **ปลั๊กอิน `YEP_QuestJournal`:**
   - JSON Path: `YEP_QuestJournal.parameters['Quest Category Window']['Available Text']`
   - Original: `\i[192]Available (%1)`
   - Role: `UI/Plugin`
2. **ปลั๊กอิน `DK_Game_Time`:**
   - JSON Path: `DK_Game_Time.parameters.morningPeriod.name`
   - Original: `Morning`
   - Role: `UI/Plugin`
3. **ปลั๊กอิน `SRD_FullscreenToggleOption`:**
   - JSON Path: `SRD_FullscreenToggleOption.parameters['Option Name']`
   - Original: `Fullscreen`
   - Role: `UI/Plugin`

## 5. การจำลองการทำงานทีละขั้นตอน (Step-by-Step Execution Traces)

### ตัวอย่างที่ 1: การสกัดและการฉีดกลับของ `YEP_QuestJournal` (Nested Struct)
1. **Input ใน `plugins.js`:**
   ```json
   {"name":"YEP_QuestJournal","status":true,"parameters":{"Quest Category Window":"{\"Available Text\":\"\\\\i[192]Available (%1)\",\"Cancel Text\":\"\\\\i[161]Close\"}"}}
   ```
2. **ขั้นตอนการสกัด (Extraction):**
   - `PluginASTParser`: แปลงสตริงเป็น Object
   - `RecursiveJsonUnpacker`: ตรวจพบว่า `"Quest Category Window"` เป็น JSON String $\to$ ทำการ `json.loads()`
   - `SemanticParameterClassifier`:
     - วิเคราะห์คีย์ `"Available Text"` $\to$ ตรงกับ Whitelist Keyword $\to$ **ACCEPT**
     - วิเคราะห์คีย์ `"Cancel Text"` $\to$ ตรงกับ Whitelist Keyword $\to$ **ACCEPT**
   - `ATFPluginCSVBridge`: เขียนลง CSV
     - Row 1: `Original: "\i[192]Available (%1)"`, `Note: "YEP_QuestJournal.parameters['Quest Category Window']['Available Text']"`
     - Row 2: `Original: "\i[161]Close"`, `Note: "YEP_QuestJournal.parameters['Quest Category Window']['Cancel Text']"`
3. **ขั้นตอนการแปล (Through ATF Pipeline):**
   - คำแปล Row 1: `"\i[192]เควสต์ที่รับได้ (%1)"`
   - คำแปล Row 2: `"\i[161]ปิดหน้าต่าง"`
4. **ขั้นตอนการฉีดกลับ (Injection):**
   - `RecursiveJsonPacker`: นำคำแปลภาษาไทยไปแทนที่ใน Struct ย่อย
   - `json.dumps()` Struct ย่อย $\to$ สร้างสตริงที่มี Escape Slashes ที่ถูกต้องอัตโนมัติ:
     `"{\"Available Text\":\"\\\\i[192]เควสต์ที่รับได้ (%1)\",\"Cancel Text\":\"\\\\i[161]ปิดหน้าต่าง\"}"`
   - บันทึกกลับลง `plugins.js` ผ่าน Atomic Persistence

## 6. อัลกอริทึมการตรวจสอบความถูกต้องของแท็กควบคุม (Tag Parity Verifier)

ก่อนที่จะยอมให้ฉีดคำแปลภาษาไทยกลับลงสู่ `plugins.js` สคริปต์จะต้องตรวจสอบความครบถ้วนของแท็กควบคุม:
```python
def verify_control_tag_parity(orig: str, trans: str) -> Tuple[bool, str]:
    # 1. ตรวจสอบ RPG Maker Format Specifiers (%1, %2, ...)
    orig_formats = set(re.findall(r"%\d+", orig))
    trans_formats = set(re.findall(r"%\d+", trans))
    if orig_formats != trans_formats:
        return False, f"Format specifier mismatch: expected {orig_formats}, got {trans_formats}"

    # 2. ตรวจสอบ Icon & Color Tags (\i[n], \c[n])
    orig_tags = set(re.findall(r"\\[iIcC]\[\d+\]", orig))
    trans_tags = set(re.findall(r"\\[iIcC]\[\d+\]", trans))
    if orig_tags != trans_tags:
        return False, f"Control tag mismatch: expected {orig_tags}, got {trans_tags}"

    return True, "Parity Verified"
```
